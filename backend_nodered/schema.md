CREATE TABLE public.events (
    id bigint GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    created_at timestamp with time zone DEFAULT timezone('utc'::text, now()) NOT NULL,
    event_type text NOT NULL CHECK (event_type IN ('button_press', 'motion_detected', 'snapshot')),
    image_url text, -- URL ảnh đơn (button_press, snapshot)
    images text[], -- Array URLs cho burst mode (motion_detected: 3 ảnh)
    description text NOT NULL, -- Mô tả sự kiện
    is_read boolean DEFAULT false, -- Đã xem chưa
    metadata jsonb -- Thông tin bổ sung (ví dụ: PIR sensor status)
);

CREATE TABLE public.voice_notes (
    id bigint GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    created_at timestamp with time zone DEFAULT timezone('utc'::text, now()) NOT NULL,
    audio_url text NOT NULL, -- URL file .wav từ bell-audio bucket
    duration_seconds int CHECK (duration_seconds > 0 AND duration_seconds <= 60), -- Giới hạn 60s
    transcription text, -- Text transcript (future feature)
    is_listened boolean DEFAULT false, -- Đã nghe chưa
    metadata jsonb -- Thông tin bổ sung (sample rate, etc.)
);

CREATE TABLE public.sensor_logs (
    id bigint GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    created_at timestamp with time zone DEFAULT timezone('utc'::text, now()) NOT NULL,
    sensor_type text DEFAULT 'temperature' CHECK (sensor_type IN ('temperature', 'humidity')),
    temperature float CHECK (temperature >= -40 AND temperature <= 80), -- DHT22 range
    humidity float CHECK (humidity >= 0 AND humidity <= 100), -- % RH
    metadata jsonb -- Thông tin bổ sung (sensor ID, location, etc.)
);


CREATE TABLE public.device_settings (
    id int PRIMARY KEY DEFAULT 1 CHECK (id = 1), -- Chỉ cho phép 1 row
    updated_at timestamp with time zone DEFAULT timezone('utc'::text, now()) NOT NULL,
    
    -- Chức năng còi báo động (Flow 2: PIR trigger)
    alarm_enabled boolean DEFAULT false, 
    
    -- Chế độ không làm phiền (tắt push notification đến mobile)
    do_not_disturb boolean DEFAULT false,
    
    -- Âm lượng loa ESP32 (0-100%)
    speaker_volume int DEFAULT 80 CHECK (speaker_volume >= 0 AND speaker_volume <= 100),
    
    -- Bật/tắt cảm biến PIR (để bảo trì)
    pir_enabled boolean DEFAULT true,
    
    -- Bật/tắt notification qua Web/Mobile
    notifications_enabled boolean DEFAULT true,
    
    -- Thông tin phiên bản firmware (để kiểm tra update)
    firmware_version text,
    
    -- Last sync timestamp
    last_sync timestamp with time zone,
    
    -- Metadata cho các setting khác
    metadata jsonb
);

-- Trigger để tự động cập nhật updated_at
CREATE OR REPLACE FUNCTION update_updated_at_column()
RETURNS TRIGGER AS $$
BEGIN
    NEW.updated_at = timezone('utc'::text, now());
    RETURN NEW;
END;
$$ language 'plpgsql';

CREATE TRIGGER update_device_settings_updated_at 
BEFORE UPDATE ON device_settings
FOR EACH ROW EXECUTE FUNCTION update_updated_at_column();

CREATE TABLE public.quick_responses (
    id bigint GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    created_at timestamp with time zone DEFAULT timezone('utc'::text, now()),
    title text NOT NULL, -- Tên nút hiển thị trên Web (ví dụ: "Đợi 5 phút")
    audio_url text NOT NULL, -- Link file âm thanh trong Storage
    is_active boolean DEFAULT true -- Ẩn hiện nếu cần
);

create table public.email_recipients (
  id bigint generated by default as identity not null,
  created_at timestamp with time zone not null default timezone('utc'::text, now()),
  email text not null,
  name text null,
  is_active boolean not null default true,
  notification_types jsonb not null default '{"button": true, "pir": true, "pir_alert": true, "voice": true}'::jsonb,
  constraint email_recipients_pkey primary key (id),
  constraint email_recipients_email_key unique (email),
  constraint email_recipients_email_check check ((email ~* '^[A-Za-z0-9._%+-]+@[A-Za-z0-9.-]+\.[A-Za-z]{2,}$'::text))
) tablespace pg_default;
